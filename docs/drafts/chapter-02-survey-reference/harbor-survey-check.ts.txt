import assert from 'node:assert/strict';

for(const order of [[0,1],[1,0]]){
 const s=new HarborSurvey();assert.equal(s.pending,'L1-S01');assert.ok(s.tick(26).includes('摄像仪匣'));assert.equal(s.tick(26),'');

 assert.equal(photo(s,0,false),false);assert.equal(photo(s,0,true,false),false);assert.equal(s.photos.length,0);

 s.operate('valve-s');assert.equal(s.valve,false);photo(s,'manual');s.operate('valve-c');assert.equal(s.valve,false);s.say('已经关好了');assert.equal(s.reported,false);
 s.operate('valve-s');s.say('没关好，等一下');assert.equal(s.reported,false);s.tick(61);assert.equal(s.pending,'L1-S04');s.accept('L1-S04');assert.equal(s.pending,null);
